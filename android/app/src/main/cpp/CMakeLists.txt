cmake_minimum_required(VERSION 3.22.1)
project(demobenchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(THIRD "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party")
set(SR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../source/SecondReality")
add_subdirectory("${THIRD}/oboe" "${CMAKE_CURRENT_BINARY_DIR}/oboe")
file(GLOB_RECURSE SR_SOURCES "${SR}/*.cpp")
list(FILTER SR_SOURCES EXCLUDE REGEX "/(main|BenchHost)\\.cpp$")
list(FILTER SR_SOURCES EXCLUDE REGEX "/Graphics/Graphics\\.cpp$")
list(FILTER SR_SOURCES EXCLUDE REGEX "/Parts/MENU\\.cpp$")
list(FILTER SR_SOURCES EXCLUDE REGEX "/Music/(audioPlayer|st3play)\\.cpp$")
file(READ "${SR}/Music/st3play.cpp" TRACKER)
set(TRACKER "#include <mutex>\nextern std::recursive_mutex srAudioMutex;\n${TRACKER}")
foreach(SIGNATURE "bool FillAudioBuffer(int16_t * audioBuffer, int32_t samples)" "void GetOrderRowAndFrame(unsigned short * orderPtr, unsigned short * rowPtr, unsigned int * framePtr)")
    string(REPLACE "${SIGNATURE}\n    {" "${SIGNATURE}\n    {\n        std::lock_guard<std::recursive_mutex> guard(srAudioMutex);" TRACKER "${TRACKER}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/st3play_android.cpp" "${TRACKER}")
add_library(demobenchmark SHARED Engine.cpp Scene.cpp Gles.cpp Vulkan.cpp Audio.cpp SecondRealityAndroid.cpp "${CMAKE_CURRENT_BINARY_DIR}/st3play_android.cpp" ${SR_SOURCES})
target_include_directories(demobenchmark PRIVATE "${THIRD}" "${SR}" "${SR}/Music")
target_compile_options(demobenchmark PRIVATE -O2 -fwrapv -fno-strict-aliasing -ffp-contract=off -Wno-narrowing -Wno-int-to-pointer-cast -Wno-deprecated-declarations)
target_link_options(demobenchmark PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(demobenchmark PRIVATE oboe android log EGL GLESv3 vulkan dl)
